# Thread
```c++
#include <thread>
#include <iostream>

bool s_Finished = false;

void DoWork()
{
    //引入的时间字面量命名空间。作用：允许直接书写 1s、100ms、2h 这样的字面量，而不需要写 std::chrono::seconds(1)。常见可用后缀：h（小时）、min（分钟）、s（秒）、ms（毫秒）、us（微秒）、ns（纳秒）。
   using namespace std::literals::chrono_literals;
    while (!s_Finished)
    {
        std::cout << "Working...\n";
        std::this_thread::sleep_for(1s); //使用std::this_thread给当前线程下达命令，等待一秒
    }   
}

int main()
{
    std::thread Worker(DoWork); //启动线程

    std::cin.get();
    s_Finished = true;

    Worker.join(); //在当前线程等待worker线程结束

    std::cin.get();
}
```
