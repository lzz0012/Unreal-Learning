## 单例模式
```c++
class Random
{
private:
	Random();
	float IFloat(){return m_RandomGenerator;}
public:
	Random(const Random&) = delete;

	static Random& GetInstance()
	{
		static Random instance;
		return instance;
	}
	
	static float Float() { return GetInstance().IFloat(); }

private:
	float m_RandomGenerator = 0.5f;
};
```
